.options savecurrents
.control
    save all

    * =========================================
    * CASE 1: Vd = 0 V  --> Vsd = 1.8 V
    * =========================================
    alter Vd 0
    dc Vg 1.8 0 -0.01

    let id = abs(Vd#branch)

    * IMPORTANT: deriv() is still versus Vg here
    let gm = -deriv(id)

    let gm_id = gm / id
    let id_w = id / 10e-6

    * NOW change x-axis to Id/W
    setscale id_w


    * =========================================
    * CASE 2: Vd = 0.9 V --> Vsd = 0.9 V
    * =========================================
    alter Vd 0.9
    dc Vg 1.8 0 -0.01

    let id = abs(Vd#branch)

    let gm = -deriv(id)

    let gm_id = gm / id
    let id_w = id / 10e-6

    setscale id_w


    * =========================================
    * CASE 3: Vd = 1.2 V --> Vsd = 0.6 V
    * =========================================
    alter Vd 1.2
    dc Vg 1.8 0 -0.01

    let id = abs(Vd#branch)

    let gm = -deriv(id)

    let gm_id = gm / id
    let id_w = id / 10e-6

    setscale id_w

    * =========================================
    * CASE 4: Vd = 1.5 V --> Vsd = 0.3 V
    * =========================================
    alter Vd 1.5
    dc Vg 1.8 0 -0.01

    let id = abs(Vd#branch)

    let gm = -deriv(id)

    let gm_id = gm / id
    let id_w = id / 10e-6

    setscale id_w

    * Friendly legends
    let VSD_1v8 = dc1.gm_id
    let VSD_0v9 = dc2.gm_id
    let VSD_0v6 = dc3.gm_id
    let VSD_0v3 = dc4.gm_id


    * Confirm that both simulations exist
    setplot

    * Overlay them
    plot VSD_1v8 VSD_0v9 VSD_0v6 VSD_0v3 xlog

.endc














.options savecurrents

.control
    save all

    * =====================================================
    * PMOS gm/Id characterization
    *
    * VS = VB = 1.8 V
    * W  = 10 um
    * L  = 0.5 um
    *
    * Sweep VG from 1.8 V -> 0 V
    * Compare VSD = 1.8, 0.9, 0.6, 0.3 V
    * =====================================================


    * -----------------------------------------------------
    * CASE 1: VD = 0 V
    *         VSD = 1.8 - 0 = 1.8 V
    * -----------------------------------------------------
    alter Vd 0
    dc Vg 1.8 0 -0.001

    let id = abs(Vd#branch)

    * Since the sweep variable is VG:
    * gm = dId/dVSG = -dId/dVG
    let gm = -deriv(id)

    let gm_id = gm / id
    let id_w  = id / 10e-6


    * -----------------------------------------------------
    * CASE 2: VD = 0.9 V
    *         VSD = 1.8 - 0.9 = 0.9 V
    * -----------------------------------------------------
    alter Vd 0.9
    dc Vg 1.8 0 -0.001

    let id = abs(Vd#branch)
    let gm = -deriv(id)

    let gm_id = gm / id
    let id_w  = id / 10e-6


    * -----------------------------------------------------
    * CASE 3: VD = 1.2 V
    *         VSD = 1.8 - 1.2 = 0.6 V
    * -----------------------------------------------------
    alter Vd 1.2
    dc Vg 1.8 0 -0.001

    let id = abs(Vd#branch)
    let gm = -deriv(id)

    let gm_id = gm / id
    let id_w  = id / 10e-6


    * -----------------------------------------------------
    * CASE 4: VD = 1.5 V
    *         VSD = 1.8 - 1.5 = 0.3 V
    * -----------------------------------------------------
    alter Vd 1.5
    dc Vg 1.8 0 -0.001

    let id = abs(Vd#branch)
    let gm = -deriv(id)

    let gm_id = gm / id
    let id_w  = id / 10e-6


    * =====================================================
    * Copy results into the constant plot so that the
    * vectors have useful names in the plot legend.
    * =====================================================
    setplot const

    let vsd_1v8 = dc1.gm_id
    let idw_1v8 = dc1.id_w

    let vsd_0v9 = dc2.gm_id
    let idw_0v9 = dc2.id_w

    let vsd_0v6 = dc3.gm_id
    let idw_0v6 = dc3.id_w

    let vsd_0v3 = dc4.gm_id
    let idw_0v3 = dc4.id_w


    * =====================================================
    * Plot gm/Id vs Id/W
    * Each Y curve explicitly gets its own X vector.
    * =====================================================
    plot \
        vsd_1v8 vs idw_1v8 \
        vsd_0v9 vs idw_0v9 \
        vsd_0v6 vs idw_0v6 \
        vsd_0v3 vs idw_0v3 \
        xlog \
        xlabel 'Id/W [uA/um]' \
        ylabel 'gm/Id [1/V]' \
        title 'PMOS gm/Id Characterization'

.endc
